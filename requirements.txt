# Runtime dependencies. Keep this minimal — it is what Horizon installs.

# plain Python functions in ~46 places; that behaviour is version-dependent, so
# an unpinned range could break the server on a rebuild.

httpx>=0.27,<1.0

# Only imported when USE_OS_TRUSTSTORE=1 (see server.py). Needed on corporate

# not trust. Unnecessary on a cloud host, but harmless and cheap to install.
truststore>=0.9
